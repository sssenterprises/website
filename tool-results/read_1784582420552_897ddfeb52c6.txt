   522→// ─── Admin Dashboard ─────────────────────────────────────────────────────────
   523→
   524→function AdminDashboard({ onLogout }: { onLogout: () => void }) {
   525→  const { navigate } = useNavigation();
   526→  const [products, setProducts] = useState<Product[]>([]);
   527→  const [loading, setLoading] = useState(true);
   528→  const [search, setSearch] = useState("");
   529→  const [brandFilter, setBrandFilter] = useState("all");
   530→  const [showForm, setShowForm] = useState(false);
   531→  const [editingProduct, setEditingProduct] = useState<Product | null>(null);
   532→  const [deletingProduct, setDeletingProduct] = useState<Product | null>(null);
   533→  const [toast, setToast] = useState("");
   534→
   535→  const fetchProducts = useCallback(async () => {
   536→    try {
   537→      const params = new URLSearchParams();
   538→      if (brandFilter !== "all") params.set("brand", brandFilter);
   539→      if (search) params.set("search", search);
   540→
   541→      const res = await fetch(`/api/admin/products?${params.toString()}`);
   542→      const data = await res.json();
   543→      if (data.success) {
   544→        setProducts(data.products);
   545→      }
   546→    } catch (error) {
   547→      console.error("Error fetching products:", error);
   548→    } finally {
   549→      setLoading(false);
   550→    }
   551→  }, [brandFilter, search]);
   552→
   553→  useEffect(() => {
   554→    fetchProducts();
   555→  }, [fetchProducts]);
   556→
   557→  const showToast = (message: string) => {
   558→    setToast(message);
   559→    setTimeout(() => setToast(""), 3000);
   560→  };
   561→
   562→  const handleSave = async (data: ProductFormData) => {
   563→    try {
   564→      if (editingProduct) {
   565→        // Update
   566→        const res = await fetch(`/api/products/${editingProduct.id}`, {
   567→          method: "PUT",
   568→          headers: { "Content-Type": "application/json" },
   569→          body: JSON.stringify(data),
   570→        });
   571→        const result = await res.json();
   572→        if (result.success) {
   573→          showToast("Product updated successfully!");
   574→        } else {
   575→          showToast("Failed to update product.");
   576→        }
   577→      } else {
   578→        // Create
   579→        const res = await fetch("/api/products", {
   580→          method: "POST",
   581→          headers: { "Content-Type": "application/json" },
   582→          body: JSON.stringify(data),
   583→        });
   584→        const result = await res.json();
   585→        if (result.success) {
   586→          showToast("Product added successfully!");
   587→        } else {
   588→          showToast(result.message || "Failed to add product.");
   589→        }
   590→      }
   591→      setShowForm(false);
   592→      setEditingProduct(null);
